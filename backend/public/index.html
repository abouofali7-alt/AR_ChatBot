POST /api/webhook (No auth)

curl -X POST https://archatbot-production.up.railway.app/api/webhook \
  -H "Content-Type: application/json" \
  -d '{"message":"Hello","language":"en"}'

POST /api/chat/stream (Streaming)

curl -X POST https://archatbot-production.up.railway.app/api/chat/stream \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"message":"Hello","systemPrompt":"You are helpful"}' \
  --no-buffer

GET /api/conversations

curl https://archatbot-production.up.railway.app/api/conversations \
  -H "Authorization: Bearer YOUR_API_KEY"

POST /api/conversations

curl -X POST https://archatbot-production.up.railway.app/api/conversations \

  -H "Content-Type: application/json" \
  -d '{"title":"My Chat"}'
